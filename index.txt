  return '₹' + n.toLocaleString('en-IN', {maximumFractionDigits:0});
}

/* ---------------- AI classifier ---------------- */
function classify(text){
  const lower = text.toLowerCase();
  let type = 'Expense';
  if (INCOME_HINTS.some(k=>lower.includes(k))) type = 'Income';
  if (EXPENSE_HINTS.some(k=>lower.includes(k))) type = 'Expense';

  let category = 'Other';
  for (const [cat, keys] of Object.entries(CATEGORY_KEYWORDS)){
    if (keys.some(k=>lower.includes(k))){ category = cat; break; }
  }
  if (type==='Income' && category!=='Sales' && category==='Other') category='Sales';

  const amtMatch = lower.replace(/,/g,'').match(/(?:₹|rs\.?|inr)?\s?(\d+(\.\d+)?)/);
  const amount = amtMatch ? parseFloat(amtMatch[1]) : 0;

  const nature = type==='Income' ? 'Revenue transaction' : 'Business expense';
  return { nature, type, category, amount };
}

}

function calcInvoice(){
  const rows = [...document.querySelectorAll('#invItemsBody tr')];
  if (!rows.length){ toast('Add at least one item'); return; }
  let subtotal=0, discount=0;
  const items = rows.map(r=>{
    const name = r.querySelector('.it-name').value || 'Item';
    const qty = parseFloat(r.querySelector('.it-qty').value)||0;
    const price = parseFloat(r.querySelector('.it-price').value)||0;
    const disc = parseFloat(r.querySelector('.it-disc').value)||0;
    const lineAmt = qty*price - disc;
    subtotal += qty*price; discount += disc;
    return {name, qty, amount:lineAmt};
  });
  const taxable = subtotal-discount;
  const rate = parseFloat(document.getElementById('invGstRate').value);
  const taxType = document.getElementById('invTaxType').value;
  const tax = taxable*rate/100;
  const grand = taxable+tax;

  document.getElementById('pvNumber').textContent = document.getElementById('invNumber').value || 'INV-1001';
  document.getElementById('pvDate').textContent = document.getElementById('invDate').value || todayStr();
  document.getElementById('pvCustomer').textContent = document.getElementById('invCustName').value || '—';
  document.getElementById('pvItemsTable').querySelector('tbody').innerHTML =
    items.map(i=>`<tr><td>${i.name}</td><td>${i.qty}</td><td>${fmt(i.amount)}</td></tr>`).join('');
  document.getElementById('pvSubtotal').textContent = fmt(subtotal);
  document.getElementById('pvDiscount').textContent = fmt(discount);
  document.getElementById('pvTaxable').textContent = fmt(taxable);
  document.getElementById('pvTaxLabel').textContent = taxType==='split' ? 'CGST + SGST' : 'IGST';
  document.getElementById('pvTax').textContent = fmt(tax);
  document.getElementById('pvGrand').textContent = fmt(grand);
  document.getElementById('invPreviewBizName').textContent = document.getElementById('invBizName').value || 'Your Business';

  window._currentInvoice = {items, subtotal, discount, taxable, rate, taxType, tax, grand,
    number:document.getElementById('invNumber').value, date:document.getElementById('invDate').value||todayStr(),
    customer:document.getElementById('invCustName').value||'Walk-in Customer'};
}

function renderReports(){
  const period = document.getElementById('reportPeriod').value;
  const now = new Date();
  const inRange = (dateStr)=>{
    if (period==='all') return true;
    const d = new Date(dateStr);
    if (isNaN(d)) return true;
    if (period==='daily') return dateStr===todayStr();
    if (period==='monthly') return d.getFullYear()===now.getFullYear() && d.getMonth()===now.getMonth();
    if (period==='quarterly') return d.getFullYear()===now.getFullYear() && Math.floor(d.getMonth()/3)===Math.floor(now.getMonth()/3);
    if (period==='yearly') return d.getFullYear()===now.getFullYear();
    return true;
  };
  const list = state.transactions.filter(t=>inRange(t.date));
  const rev = list.filter(t=>t.type==='Income').reduce((s,t)=>s+t.amount,0);
  const exp = list.filter(t=>t.type==='Expense').reduce((s,t)=>s+t.amount,0);

  document.getElementById('repInflow').textContent = fmt(rev);
  document.getElementById('repOutflow').textContent = fmt(exp);
  document.getElementById('repNetCash').textContent = fmt(rev-exp);

  const sales = list.filter(t=>t.type==='Income');
